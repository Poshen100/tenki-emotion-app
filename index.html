`; return app; } // ============================================ // Router // ============================================ function renderPage(page: 'scan' | 'dashboard' | 'timer'): void { const container = document.querySelector('#page-container'); if (!container) return; // Clear existing container.innerHTML = ''; // Render new page let pageElement: HTMLElement; switch (page) { case 'scan': pageElement = createScanPage(); break; case 'dashboard': pageElement = createDashboardPage(); break; case 'timer': pageElement = createTimerPage(); break; default: pageElement = createScanPage(); } container.appendChild(pageElement); } // ============================================ // Keyboard Shortcuts (Debug Mode) // ============================================ function setupKeyboardShortcuts(): void { document.addEventListener('keydown', async (e) => { // Ctrl+Shift+D - Toggle debug mode if (e.ctrlKey && e.shiftKey && e.key === 'D') { e.preventDefault(); store.toggleDebugMode(); console.log('Debug mode:', store.getState().debugMode); } // Ctrl+Shift+E - Export ZIP if (e.ctrlKey && e.shiftKey && e.key === 'E') { e.preventDefault(); console.log('Exporting data...'); await downloadExport(); console.log('Export complete'); } // Ctrl+Shift+C - Copy snapshot if (e.ctrlKey && e.shiftKey && e.key === 'C') { e.preventDefault(); const state = store.getState(); await copySnapshot({ timestamp: new Date().toISOString(), sqs: state.currentSQS, metrics: state.currentMetrics, prs: state.currentPRs, tei: state.currentTEI }); console.log('Snapshot copied to clipboard'); } }); } // ============================================ // Service Worker Registration // ============================================ async function registerServiceWorker(): Promise { if ('serviceWorker' in navigator) { try { const registration = await navigator.serviceWorker.register('/sw.js'); console.log('SW registered:', registration.scope); } catch (error) { console.log('SW registration failed:', error); } } } // ============================================ // Initialize App // ============================================ async function initApp(): Promise { // Create app shell const app = createAppShell(); document.querySelector('#app')!.appendChild(app); // Load historical data try { const [bins, sessions] = await Promise.all([ getDerivedBins(), getSessions(20) ]); store.setDerivedBins(bins); store.setRecentSessions(sessions); } catch (error) { console.warn('Failed to load historical data:', error); } // Subscribe to page changes store.subscribe((state) => { renderPage(state.currentPage); }); // Initial render renderPage(store.getState().currentPage); } // Start the app initApp().catch(console.error);
